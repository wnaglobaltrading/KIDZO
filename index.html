function densifyPolyline(poly, step = 12) { if (!poly || poly.length < 2) return poly || []; const dense = [poly[0]]; for (let i = 1; i < poly.length; i++) { const [x0, y0] = poly[i - 1]; const [x1, y1] = poly[i]; const segLen = Math.hypot(x1 - x0, y1 - y0); const count = Math.max(1, Math.ceil(segLen / step)); for (let k = 1; k <= count; k++) { const t = k / count; dense.push([x0 + (x1 - x0) * t, y0 + (y1 - y0) * t]); } } return dense; } function distToPolylineOrdered(pt, poly, expectedIdx, windowRadius = 10) { let minD = Infinity; let bestIdx = expectedIdx; const lo = Math.max(0, expectedIdx - 3); const hi = Math.min(poly.length - 1, expectedIdx + windowRadius); for (let i = lo; i <= hi; i++) { const d = Math.hypot(pt.x - poly[i][0], pt.y - poly[i][1]); if (d < minD) { minD = d; bestIdx = i; } } return { dist: minD, index: bestIdx }; } function evaluateStrokeAttempt(strokeObj, samples) { const pts = densifyPolyline(strokeObj.pts, 12); if (!samples || samples.length < 4) { return { ok: false, reason: "short" }; } // 1. Mesti bermula berhampiran titik mula bernombor const startPt = { x: pts[0][0], y: pts[0][1] }; const firstDist = Math.hypot(samples[0].x - startPt.x, samples[0].y - startPt.y); if (firstDist > 64) { return { ok: false, reason: "start" }; } // 2. Semak kemajuan berurutan (ordered progress) supaya tidak bertindan / melompat const hitFlags = new Array(pts.length).fill(false); let cursorIdx = 0; let validSamples = 0; let totalPathDist = 0; for (let i = 0; i < samples.length; i++) { if (i > 0) { totalPathDist += Math.hypot(samples[i].x - samples[i - 1].x, samples[i].y - samples[i - 1].y); } const res = distToPolylineOrdered(samples[i], pts, cursorIdx, Math.max(10, Math.ceil(pts.length * 0.35))); if (res.dist <= 64) { validSamples += 1; if (res.index >= cursorIdx) { for (let k = cursorIdx; k <= res.index; k++) hitFlags[k] = true; cursorIdx = res.index; } hitFlags[res.index] = true; } } const hitRatio = hitFlags.filter(Boolean).length / pts.length; const sampleAccuracy = validSamples / samples.length; const reachedEnd = cursorIdx >= Math.floor((pts.length - 1) * 0.80); let expectedLen = 0; for (let i = 1; i < pts.length; i++) { expectedLen += Math.hypot(pts[i][0] - pts[i - 1][0], pts[i][1] - pts[i - 1][1]); } const lengthOk = totalPathDist >= expectedLen * 0.65 && totalPathDist <= expectedLen * 2.2; if (sampleAccuracy >= 0.72 && hitRatio >= 0.78 && reachedEnd && lengthOk) { return { ok: true }; } return { ok: false, reason: "path" }; }
